CREATE TABLE usuario(

    email varchar(50) NOT NULL,
    senha varchar(50) NOT NULL,

	id 		INTEGER				AUTO_INCREMENT	NOT NULL,
    nome		VARCHAR(50)			NOT NULL,
    genero	VARCHAR(50)			NOT NULL,
    duracao	 varchar(100)			NOT NULL,

);
